--- 5. FILES ---

git rm -r "folder"                               delete folder
git rm --cached "file"                           untrack but keep locally (add to .gitignore)
git mv "old" "new"                               move/rename
git restore --staged "file"                      unstage
git restore "file"                               undo uncommitted edits to file
